const MAX_POKEMON_LEVEL=70; // Confirmed in official Ver.3.6.0 notice, 2026-06-25. const SUBSKILL_LEVELS=[10,25,50,70,80]; // Individual corrections are deliberately separate from daily yield (energy, collection and camp are not fixed here). const NATURE_ROWS=window.PS_CATALOG.natures.rows; const NATURE_AXES=window.PS_CATALOG.natures.axes; const NATURE_EFFECTS=Object.fromEntries(NATURE_ROWS.flatMap((row,up)=>row.map((name,down)=>[name,up===down?{}:{up:NATURE_AXES[up],down:NATURE_AXES[down]}]))); const SUBSKILLS=Object.fromEntries(window.PS_CATALOG.subskills.map(skill=>[skill.name,skill])); const SUBSKILL_NAMES=window.PS_CATALOG.subskills.map(skill=>skill.name); function individualCorrections(item,p,teamSpeedReduction=0){ const effect=item.mint?{}:NATURE_EFFECTS[item.nature]||{}; const active=SUBSKILL_LEVELS.filter(n=>n<=item.level).map((_,i)=>item.subskills?.[i]).filter(Boolean); const sum=kind=>active.reduce((total,name)=>total+(SUBSKILLS[name]?.effect===kind?SUBSKILLS[name].value||0:0),0); const natureFactor=axis=>effect.up===axis?window.PS_CATALOG.natures.factors[axis].up:effect.down===axis?window.PS_CATALOG.natures.factors[axis].down:1; const soloSpeedReduction=Math.min(.35,sum('speedReduction')); const speedReduction=Math.min(.35,sum('speedReduction')+teamSpeedReduction); const speedBase=Number.isFinite(p?.help)?p.help*(1-(item.level-1)*.002)*natureFactor('speed'):null; const soloSpeed=speedBase==null?null:Math.floor(speedBase*(1-soloSpeedReduction)); const speed=speedBase==null?null:Math.floor(speedBase*(1-speedReduction)); const foodFactor=natureFactor('food')*(1+sum('foodRateBonus')); const skillFactor=natureFactor('skill')*(1+sum('skillRateBonus')); const carry=Number.isFinite(p?.carry)?p.carry+sum('carryBonus'):null; const food=Number.isFinite(p?.foodRate)?Math.min(100,p.foodRate*foodFactor):null; const skill=Number.isFinite(p?.skillRate)?Math.min(100,p.skillRate*skillFactor):null; const unknown=[...active.filter(name=>!SUBSKILL_NAMES.includes(name)),...(!item.mint&&item.nature&&!NATURE_EFFECTS[item.nature]?[item.nature]:[])]; return {speed,soloSpeed,carry,food,skill,foodFactor,skillFactor,active,unknown, nature:effect,berryQty:Number.isFinite(p?.berryQty)?p.berryQty+sum('berryQtyBonus'):null, teamBonus:sum('teamSpeedReduction')>0,energyFactor:natureFactor('energy'),expFactor:natureFactor('exp')}; } function teamSpeedContext(ids=team,box=state.box,catalog=window.PS_CATALOG){ const selected=ids.map(id=>box.find(item=>item.id===id)).filter(Boolean); const contribution=item=>SUBSKILL_LEVELS.reduce((sum,unlock,i)=>sum+(unlock<=item.level&&SUBSKILLS[item.subskills?.[i]]?.effect==='teamSpeedReduction'?SUBSKILLS[item.subskills[i]].value:0),0); const bonus=selected.reduce((sum,item)=>sum+contribution(item),0); const members=new Map(selected.map(item=>[item.id,individualCorrections(item,catalog?.pokemon?.[item.no],bonus)])); return {bonus,bonusCount:selected.filter(item=>contribution(item)>0).length,members}; }

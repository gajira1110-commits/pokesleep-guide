return stockLimit===2?2-(2+lambda)*none:1-none; } function berryEnergyAtLevel(base,level){ if(!Number.isFinite(base)||!Number.isFinite(level))return null; return Math.round(Math.max(base+level-1,base*Math.pow(1.025,level-1)));

// Three hours at the adopted maximum-energy rate. This never invokes daily skills. function whistleEstimate(ids,box,catalog,options={}){ const selected=ids.filter(Boolean).map(id=>box.find(item=>item.id===id)); const empty={members:[],foods:new Map(),berryEnergy:null,berryCount:null,pendingReasons:[]}; if(!selected.length)return empty; if(selected.some(item=>!item))return {...empty,pendingReasons:['チーム個体が見つかりません']}; if(options.fieldMode==='ex')return {...empty,pendingReasons:['EXフィールドの補正は未対応']}; const speedContext=teamSpeedContext(ids,box,catalog),members=[],pendingReasons=[]; if(Number.isFinite(options.teamSpeedReduction))for(const item of selected)speedContext.members.set(item.id,individualCorrections(item,speciesFor(item,catalog),options.teamSpeedReduction)); const favorites=new Set(options.favoriteBerries||[]),areaFactor=1+(options.areaBonus||0)/100; for(const item of selected){ const species=speciesFor(item,catalog),stats=speedContext.members.get(item.id),reasons=[]; if(!species||species.mythicalSettings||species.dailyCalculationStatus==='pending_special_skill')reasons.push('特殊な個体・追加状態は未対応'); if(!Number.isFinite(item.level)||item.level<1)reasons.push('レベル'); if(!options.assumeNeutral){ if(!item.nature&&!item.mint)reasons.push('性格未入力'); if(SUBSKILL_LEVELS.some((level,index)=>level<=item.level&&!item.subskills?.[index]))reasons.push('解放済みサブスキル未入力'); } if(stats?.unknown?.length)reasons.push('性格・サブスキル未対応'); if(!Number.isFinite(stats?.speed)||stats.speed<=0)reasons.push('おてつだい時間'); if(!Number.isFinite(stats?.food)||stats.food<0||stats.food>100)reasons.push('食材確率'); if(!Number.isFinite(stats?.berryQty)||stats.berryQty<=0)reasons.push('きのみ個数'); const berryBase=catalog.berries?.[species?.berry]; if(!Number.isFinite(berryBase))reasons.push('きのみエナジー'); const slots=(species?.ingredientSlots||[]).filter(slot=>slot.unlock<=item.level).map(slot=>{ const candidate=slot.candidates?.length===1?slot.candidates[0]:slot.candidates?.find(c=>c.name===item.ingredients?.[slot.unlock]); if(!candidate||!Number.isFinite(candidate.qty)||candidate.qty<=0)reasons.push(`Lv.${slot.unlock}の食材選択・個数`); return candidate; }); if(!slots.length)reasons.push('食材枠'); if(reasons.length){pendingReasons.push(`${item.name}：${[...new Set(reasons)].join('・')}`);continue} const helps=10800/(stats.speed*.45),chance=stats.food/100; const berries=Math.floor(helps*(1-chance)*stats.berryQty+.5),foods=new Map(); // Round each slot before merging identical ingredient names or individuals. for(const slot of slots)foods.set(slot.name,(foods.get(slot.name)||0)+Math.floor(helps*chance*slot.qty/slots.length+.5)); const berryEnergy=options.favoritesKnown===false?null:berries*berryEnergyAtLevel(berryBase,item.level)*(favorites.has(species.berry)?2:1)*areaFactor; members.push({id:item.id,name:item.name,level:item.level,berryName:species.berry,berries,berryEnergy,foods,helpSeconds:stats.speed}); } // A subset is never presented as the selected team's total. if(pendingReasons.length)return {...empty,pendingReasons}; const foods=new Map();for(const member of members)for(const [name,qty] of member.foods)foods.set(name,(foods.get(name)||0)+qty); return {members,foods,berryCount:members.reduce((sum,m)=>sum+m.berries,0),berryEnergy:members.some(m=>m.berryEnergy===null)?null:members.reduce((sum,m)=>sum+m.berryEnergy,0),pendingReasons}; } // Exact additive optimization for each attainable team helping-bonus total. // Individual yields are rounded by whistleEstimate; no solo top-five heuristic. function whistleRecommend(box,catalog,options={},target='berry',maxSize=5){ const unique=[...new Map(box.map(item=>[item.id,item])).values()],excluded=[],candidates=[]; if(options.fieldMode==='ex')return {ids:[],excluded,reason:'EXフィールドの補正は未対応です。'}; if((target==='berry'||Array.isArray(target)&&target.some(t=>t.name==='berry'))&&options.favoritesKnown===false)return {ids:[],excluded,reason:'好物きのみを3種類設定してください。'}; for(const item of unique){ const result=whistleEstimate([item.id],unique,catalog,options); if(result.pendingReasons.length||!result.members.length){excluded.push(...result.pendingReasons);continue} const species=speciesFor(item,catalog); const contribution=SUBSKILL_LEVELS.reduce((sum,unlock,i)=>sum+(unlock<=item.level&&SUBSKILLS[item.subskills?.[i]]?.effect==='teamSpeedReduction'?SUBSKILLS[item.subskills[i]].value:0),0); candidates.push({item,species,bonus:Math.round(contribution*1000000)}); } const size=Math.min(maxSize,candidates.length); if(!size)return {ids:[],excluded,reason:'計算できるボックス個体がありません。'}; const sums=Array.from({length:size+1},()=>new Set());sums[0].add(0); for(const c of candidates)for(let n=size;n>0;n--)for(const sum of sums[n-1])sums[n].add(sum+c.bonus); let best=null; for(const total of sums[size]){ const dp=Array.from({length:size+1},()=>new Map());dp[0].set('0|',{score:0,bonus:0,ids:[],special:[]}); for(const c of candidates){ const result=whistleEstimate([c.item.id],unique,catalog,{...options,teamSpeedReduction:total/1000000}); if(result.pendingReasons.length)continue; const score=Array.isArray(target)?target.reduce((sum,t)=>sum+(t.name==='berry'?result.berryEnergy:result.foods.get(t.name)||0)/t.maximum*t.weight,0):target==='berry'?result.berryEnergy:(result.foods.get(target)||0); if(!Number.isFinite(score))continue; for(let n=size;n>0;n--)for(const prev of dp[n-1].values()){ const bonus=prev.bonus+c.bonus;if(bonus>total)continue; const special=c.species.specialTeamLimited===true; if(special&&!specialTeamCompatible(c.species,prev.special))continue; const nextSpecial=special?[...prev.special,c.species]:prev.special; const key=bonus+'|'+nextSpecial.map(p=>p.speciesId||p.no).sort().join(','); const next={score:prev.score+score,bonus,ids:[...prev.ids,c.item.id],special:nextSpecial}; if(!dp[n].has(key)||next.score>dp[n].get(key).score)dp[n].set(key,next); } } for(const result of dp[size].values())if(result.bonus===total&&(!best||result.score>best.score))best=result; } // Fewer than five compatible members can be legitimate in a small special-only box. if(!best&&size>1){ return whistleRecommend(unique,catalog,options,target,size-1); } return best?{ids:best.ids,result:whistleEstimate(best.ids,unique,catalog,options),excluded}:{ids:[],excluded,reason:'編成可能な個体がありません。'}; } function whistleRecommendWeighted(box,catalog,options,targets){ const selected=targets.filter(t=>t.name&&Number.isFinite(t.weight)&&t.weight>0); if(!selected.length)return {ids:[],reason:'項目を1つ以上選んでください。',excluded:[]}; if(new Set(selected.map(t=>t.name)).size!==selected.length)return {ids:[],reason:'同じ項目は重複して選べません。',excluded:[]}; const normalized=[]; for(const t of selected){const best=whistleRecommend(box,catalog,options,t.name);if(!best.ids.length)return best;const maximum=t.name==='berry'?best.result.berryEnergy:best.result.foods.get(t.name)||0;if(maximum<=0)return {ids:[],reason:`${t.name==='berry'?'きのみ':t.name}の獲得見込みがある個体がありません。`,excluded:best.excluded};normalized.push({...t,maximum})} const recommendation=whistleRecommend(box,catalog,options,normalized); recommendation.targets=normalized;return recommendation; }
